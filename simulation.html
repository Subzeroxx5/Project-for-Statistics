How it works: Each simulation injects Gaussian noise ε ~ N(0, σ²) into all 142 country values, then computes the global mean. After N runs the distribution of means approximates N(μ, σ²/n) by the Central Limit Theorem. The expected output SD = σ / √142 ≈ σ × 0.0839.

Code Appendix — Section 3 (does not count toward page limit)

import numpy as np, pandas as pd from scipy import stats # ── Load Gapminder dataset ───────────────────────────────────── df = pd.read_csv('gapminder.csv') df_2007 = df[df['year'] == 2007] countries = df_2007['lifeExp'].values # ── Descriptive statistics ───────────────────────────────────── print(df_2007['lifeExp'].describe()) print(df_2007.groupby('continent')['lifeExp'].median()) # ── Monte Carlo — Gaussian noise per country ─────────────────── np.random.seed(42) sigma_meas = 2.0 # measurement uncertainty (years) N_SIMS = 10_000 simulated_means = [] for _ in range(N_SIMS): noisy = countries + np.random.normal(0, sigma_meas, size=len(countries)) simulated_means.append(noisy.mean()) sim_arr = np.array(simulated_means) ci_90 = np.percentile(sim_arr, [5, 95]) print(f"True mean : {countries.mean():.2f}") print(f"Sim mean : {sim_arr.mean():.2f}") print(f"Sim std : {sim_arr.std():.3f}") # ≈ sigma / sqrt(n) print(f"90% CI : [{ci_90[0]:.2f}, {ci_90[1]:.2f}]") # ── 2030 Scenario distributions ──────────────────────────────── scenarios = { 'Optimistic' : (76.5, 2.2), 'Baseline' : (72.0, 3.1), 'Pessimistic': (67.5, 4.8), } x = np.linspace(50, 95, 500) for name, (mu, sd) in scenarios.items(): pdf = stats.norm.pdf(x, mu, sd) print(ff"{name}: 90% CI = [{stats.norm.ppf(0.05,mu,sd):.1f}, {stats.norm.ppf(0.95,mu,sd):.1f}]") # ── R equivalent ─────────────────────────────────────────────── # library(gapminder); set.seed(42) # vals <- gapminder %>% filter(year==2007) %>% pull(lifeExp) # sims <- replicate(10000, mean(vals + rnorm(length(vals), 0, 2))) # quantile(sims, c(0.05, 0.50, 0.95))
